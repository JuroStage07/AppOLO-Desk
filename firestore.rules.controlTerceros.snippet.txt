// =========================
// ✅ Control Terceros
// =========================
// El cliente (src/services/controlTerceros.js → registrarEvento) envía además
// tenantId y company para acotar por empresa; deben estar en hasOnly.
//
match /controlTerceros/{eventId} {

  function canSaludOcupacional() {
    return authed() && (isAdminRequester() || hasPerm('saludOcupacional'));
  }

  allow get, list: if canSaludOcupacional();

  allow create: if canSaludOcupacional()
    && request.resource.data.keys().hasOnly([
      'terceroId',
      'tipo',
      'dayKey',
      'createdAt',
      'creadoPorUid',
      'terceroNombre',
      'terceroCedula',
      'terceroEmpresa',
      'terceroMotivo',
      'tenantId',
      'company'
    ])
    && request.resource.data.terceroId is string
    && request.resource.data.tipo is string
    && request.resource.data.tipo in ['ENTRADA','SALIDA']
    && request.resource.data.dayKey is string
    && request.resource.data.createdAt is timestamp
    && (!('creadoPorUid' in request.resource.data) || request.resource.data.creadoPorUid is string)
    && (!('terceroNombre' in request.resource.data) || request.resource.data.terceroNombre is string)
    && (!('terceroCedula' in request.resource.data) || request.resource.data.terceroCedula is string)
    && (!('terceroEmpresa' in request.resource.data) || request.resource.data.terceroEmpresa is string)
    && (!('terceroMotivo' in request.resource.data) || request.resource.data.terceroMotivo is string)
    && (!('tenantId' in request.resource.data) || request.resource.data.tenantId is string)
    && (!('company' in request.resource.data) || request.resource.data.company is string);

  allow update, delete: if false;
}

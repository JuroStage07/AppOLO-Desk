    // =========================
    // DOCUMENTACION
    // =========================
    function canReadDocumentacionStorage(tenantId, company) {
      return isAdmin()
        || (
          sameTenantPath(tenantId, company)
          && (
            hasPerm("documentacion")
            || hasPerm("documentos")
            || role() in ["administrativo", "dev", "admin"]
          )
        );
    }

    function canWriteDocumentacionStorage(tenantId, company) {
      return sameTenantPath(tenantId, company)
        && isAdmin()
        && request.resource != null
        && request.resource.contentType == "application/pdf";
    }

    // ... en el match:

    match /documentacion/{tenantId}/{company}/{allPaths=**} {
      allow read: if canReadDocumentacionStorage(tenantId, company);
      allow write: if canWriteDocumentacionStorage(tenantId, company);
    }

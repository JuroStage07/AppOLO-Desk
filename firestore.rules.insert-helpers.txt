    function canMantenimientoOT() {
      return authed() && (
        isAdminRequester() ||
        hasPerm('mantenimiento')
      );
    }

    function validOTState(s) {
      return s is string
        && s in ['Solicitada', 'En proceso', 'En revisión', 'Finalizada'];
    }

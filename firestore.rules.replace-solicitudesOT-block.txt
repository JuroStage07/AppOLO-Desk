// ===== REEMPLAZAR por completo el bloque `match /solicitudesOT/{solicitudId} { ... }` (incluye match /subtareas anidado) por esto: =====

match /solicitudesOT/{solicitudId} {

  function canSolicitudesOT() {
    return canMantenimientoOT();
  }

  allow get, list: if canSolicitudesOT();

  allow create: if canSolicitudesOT()
    && request.resource.data.keys().hasOnly([
      'solicitanteNombre',
      'solicitanteFicha',
      'fecha',
      'nombreOT',
      'activoReferencia',
      'departamento',
      'departamentoBase',
      'departamentoOtro',
      'lugarProblema',
      'lugarProblemaBase',
      'lugarProblemaOtro',
      'tipoProblema',
      'tipoProblemaBase',
      'tipoProblemaOtro',
      'descripcionOT',
      'notas',
      'OTState',
      'NroSolicitud',
      'createdAt',
      'updatedAt',
      'createdBy',
      'createdByName',
      'subtasks'
    ])
    && request.resource.data.solicitanteNombre is string
    && request.resource.data.solicitanteFicha is string
    && request.resource.data.fecha is string
    && request.resource.data.nombreOT is string
    && request.resource.data.activoReferencia is string
    && request.resource.data.departamento is string
    && request.resource.data.departamentoBase is string
    && (!('departamentoOtro' in request.resource.data) || request.resource.data.departamentoOtro is string)
    && request.resource.data.lugarProblema is string
    && request.resource.data.lugarProblemaBase is string
    && (!('lugarProblemaOtro' in request.resource.data) || request.resource.data.lugarProblemaOtro is string)
    && request.resource.data.tipoProblema is string
    && request.resource.data.tipoProblemaBase is string
    && (!('tipoProblemaOtro' in request.resource.data) || request.resource.data.tipoProblemaOtro is string)
    && request.resource.data.descripcionOT is string
    && (!('notas' in request.resource.data) || request.resource.data.notas is string)
    && request.resource.data.OTState == 'Solicitada'
    && request.resource.data.NroSolicitud is string
    && request.resource.data.createdAt is timestamp
    && request.resource.data.updatedAt is timestamp
    && request.resource.data.createdBy == request.auth.uid
    && request.resource.data.createdByName is string
    && (!('subtasks' in request.resource.data) || request.resource.data.subtasks is list)
    && request.resource.data.nombreOT.trim().size() > 0
    && request.resource.data.activoReferencia.trim().size() > 0
    && request.resource.data.departamento.trim().size() > 0
    && request.resource.data.lugarProblema.trim().size() > 0
    && request.resource.data.tipoProblema.trim().size() > 0
    && request.resource.data.descripcionOT.trim().size() > 0;

  allow update: if canSolicitudesOT()
    && request.resource.data.diff(resource.data).changedKeys().hasOnly([
      'OTState',
      'updatedAt',
      'notas',
      'responsableUid',
      'responsableNombre'
    ])
    && (
      !request.resource.data.diff(resource.data).changedKeys().hasAny(['OTState']) ||
      validOTState(request.resource.data.OTState)
    )
    && (
      !request.resource.data.diff(resource.data).changedKeys().hasAny(['updatedAt']) ||
      request.resource.data.updatedAt is timestamp
    )
    && (
      !request.resource.data.diff(resource.data).changedKeys().hasAny(['notas']) ||
      request.resource.data.notas is string
    )
    && (
      !request.resource.data.diff(resource.data).changedKeys().hasAny(['responsableUid']) ||
      request.resource.data.responsableUid is string
    )
    && (
      !request.resource.data.diff(resource.data).changedKeys().hasAny(['responsableNombre']) ||
      request.resource.data.responsableNombre is string
    );

  allow delete: if canSolicitudesOT();

  match /subtareas/{subtareaId} {

    allow get, list: if canSolicitudesOT();

    allow create: if canSolicitudesOT()
      && request.resource.data.keys().hasOnly([
        'title',
        'description',
        'status',
        'createdAt',
        'updatedAt',
        'createdBy',
        'createdByName'
      ])
      && request.resource.data.title is string
      && request.resource.data.title.trim().size() > 0
      && (!('description' in request.resource.data) || request.resource.data.description is string)
      && request.resource.data.status is string
      && request.resource.data.createdAt is timestamp
      && request.resource.data.updatedAt is timestamp
      && request.resource.data.createdBy == request.auth.uid
      && request.resource.data.createdByName is string;

    allow update: if canSolicitudesOT()
      && request.resource.data.diff(resource.data).changedKeys().hasOnly([
        'title',
        'description',
        'status',
        'updatedAt',
        'chronoWorkAccumMs',
        'chronoDeadAccumMs',
        'chronoWorkStartedAt',
        'chronoDeadStartedAt'
      ])
      && (
        !request.resource.data.diff(resource.data).changedKeys().hasAny(['title']) ||
        (
          request.resource.data.title is string &&
          request.resource.data.title.trim().size() > 0
        )
      )
      && (
        !request.resource.data.diff(resource.data).changedKeys().hasAny(['description']) ||
        request.resource.data.description is string
      )
      && (
        !request.resource.data.diff(resource.data).changedKeys().hasAny(['status']) ||
        request.resource.data.status is string
      )
      && (
        !request.resource.data.diff(resource.data).changedKeys().hasAny(['updatedAt']) ||
        request.resource.data.updatedAt is timestamp
      )
      && (
        !request.resource.data.diff(resource.data).changedKeys().hasAny(['chronoWorkAccumMs']) ||
        (
          request.resource.data.chronoWorkAccumMs is number &&
          request.resource.data.chronoWorkAccumMs >= 0
        )
      )
      && (
        !request.resource.data.diff(resource.data).changedKeys().hasAny(['chronoDeadAccumMs']) ||
        (
          request.resource.data.chronoDeadAccumMs is number &&
          request.resource.data.chronoDeadAccumMs >= 0
        )
      )
      && (
        !request.resource.data.diff(resource.data).changedKeys().hasAny(['chronoWorkStartedAt']) ||
        !('chronoWorkStartedAt' in request.resource.data) ||
        request.resource.data.chronoWorkStartedAt is timestamp
      )
      && (
        !request.resource.data.diff(resource.data).changedKeys().hasAny(['chronoDeadStartedAt']) ||
        !('chronoDeadStartedAt' in request.resource.data) ||
        request.resource.data.chronoDeadStartedAt is timestamp
      );

    allow delete: if canSolicitudesOT();
  }
}
